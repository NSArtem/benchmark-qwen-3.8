# Reproduction commands used for this smoke test.
# All Python and package commands use the project-local uv cache because the sandbox cannot write
# the default user cache. The model runs were executed with Metal access.

mkdir -p mlx-dspark-smoke-results
cp coding_prompt.md mlx-dspark-smoke-results/coding-prompt.txt
UV_CACHE_DIR=.uv-cache uv sync
UV_CACHE_DIR=.uv-cache uv run python -m py_compile smoke_test.py
system_profiler SPHardwareDataType
sw_vers
uv --version
UV_CACHE_DIR=.uv-cache uv run python --version
UV_CACHE_DIR=.uv-cache uv run python -c 'import importlib.metadata as m; print("mlx", m.version("mlx")); print("mlx-dspark", m.version("mlx-dspark")); print("mlx-lm", m.version("mlx-lm"))'
UV_CACHE_DIR=.uv-cache uv run mlx-dspark doctor
memory_pressure
memory_pressure -Q
sysctl vm.swapusage
vm_stat

UV_CACHE_DIR=.uv-cache uv run smoke_test.py \
  --model mlx-community/Qwen3-8B-8bit --mode baseline \
  --prompt-file mlx-dspark-smoke-results/coding-prompt.txt \
  --max-new-tokens 256 --warmup-tokens 8 --kv-bits 8 \
  > mlx-dspark-smoke-results/qwen3-8b-baseline.log 2>&1

UV_CACHE_DIR=.uv-cache uv run smoke_test.py \
  --model mlx-community/Qwen3-8B-8bit --mode dspark \
  --prompt-file mlx-dspark-smoke-results/coding-prompt.txt \
  --max-new-tokens 256 --warmup-tokens 8 --kv-bits 8 \
  > mlx-dspark-smoke-results/qwen3-8b-dspark.log 2>&1

memory_pressure -Q
sysctl vm.swapusage
vm_stat
UV_CACHE_DIR=.uv-cache uv run smoke_test.py \
  --model mlx-community/Qwen3.8-27B-4bit --mode baseline \
  --prompt-file mlx-dspark-smoke-results/coding-prompt.txt \
  --max-new-tokens 256 --warmup-tokens 8 --kv-bits 8 \
  > mlx-dspark-smoke-results/qwen38-27b-baseline.log 2>&1
mv mlx-dspark-smoke-results/qwen38-27b-baseline.log \
  mlx-dspark-smoke-results/qwen38-27b-baseline-kv8-unsupported.log

memory_pressure -Q
UV_CACHE_DIR=.uv-cache uv run smoke_test.py \
  --model mlx-community/Qwen3.8-27B-4bit --mode baseline \
  --prompt-file mlx-dspark-smoke-results/coding-prompt.txt \
  --max-new-tokens 256 --warmup-tokens 8 --kv-bits 0 \
  > mlx-dspark-smoke-results/qwen38-27b-baseline.log 2>&1

# The supported 27B baseline was manually terminated after sustained paging and a prolonged
# warm-up stall; no 27B DSpark invocation was made.
memory_pressure
sysctl vm.swapusage
vm_stat
date > mlx-dspark-smoke-results/qwen38-27b-baseline-abort-observation.txt
memory_pressure >> mlx-dspark-smoke-results/qwen38-27b-baseline-abort-observation.txt
sysctl vm.swapusage >> mlx-dspark-smoke-results/qwen38-27b-baseline-abort-observation.txt
vm_stat >> mlx-dspark-smoke-results/qwen38-27b-baseline-abort-observation.txt
